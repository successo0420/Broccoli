    Usage Example
    
                
    import pickle
with open('model.pkl', 'rb') as f: model = pickle.load(f)
with open('tfidf_vectorizer.pkl', 'rb') as f: vectorizer = pickle.load(f)

print(f"Prediction: {prediction[0]}")
